set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/7f3ae3d57459e59943a4ecfefc8f6277ec6bf540 URL_HASH SHA256=7a8347c770b90b33daac2352858ca03f9c9a2ccc8ce711054870361d1a6b32e5 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "" FORCE)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS ON CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(Jolt URL https://codeload.github.com/jrouwe/JoltPhysics/tar.gz/0373ec0dd762e4bc2f6acdb08371ee84fa23c6db SOURCE_SUBDIR Build URL_HASH SHA256=834644d7e9dffee9be0220748015667649d3361611b01c7e6839a5eb58e89711 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/f5befd2d29e66809cd1110a152e375a7f1981f06 URL_HASH SHA256=85f4ce357df05bcc331b587f01976f47fb55f19fadf477a7907289686bc3f4c8 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)

FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0 URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(SDL3 Jolt imgui cgltf json stb glm assimp)
find_package(Vulkan REQUIRED)
if(NOT Vulkan_GLSLC_EXECUTABLE)
  message(FATAL_ERROR "Vulkan SDK glslc is required")
endif()
add_library(one_content STATIC src/content.cpp src/physics.cpp)
target_include_directories(one_content PUBLIC src PRIVATE ${cgltf_SOURCE_DIR} ${stb_SOURCE_DIR})
target_compile_definitions(one_content PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(one_content PUBLIC nlohmann_json::nlohmann_json Jolt glm::glm)
add_library(one_ui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
target_include_directories(one_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(one_ui PUBLIC SDL3::SDL3-static Vulkan::Vulkan)
foreach(SHADER scene.vert scene.frag shadow.vert)
  set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv")
  add_custom_command(OUTPUT "${OUTPUT}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND "${Vulkan_GLSLC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" -o "${OUTPUT}"
    DEPENDS "shaders/${SHADER}" VERBATIM)
  list(APPEND SHADERS "${OUTPUT}")
endforeach()
add_custom_target(theone_shaders DEPENDS ${SHADERS})
add_library(one_sample STATIC src/sandbox_game.cpp)
target_link_libraries(one_sample PUBLIC one_content)
add_executable(theone src/main.cpp src/renderer.cpp src/application.cpp)
target_link_libraries(theone PRIVATE one_content one_ui one_sample)
add_dependencies(theone theone_shaders)
add_custom_target(theone_runtime ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders" "$<TARGET_FILE_DIR:theone>/shaders"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:theone>/assets"
  DEPENDS theone theone_shaders VERBATIM)
if(MSVC)
  target_compile_options(theone PRIVATE /W4 /permissive- /EHsc)
  target_compile_options(one_content PRIVATE /W4 /permissive- /EHsc)
endif()
enable_testing()

target_link_libraries(one_inspect PRIVATE one_content)
add_executable(one_import tools/import_asset.cpp)
target_link_libraries(one_import PRIVATE one_content assimp::assimp)
add_executable(theone_core_tests tests/core_tests.cpp)
target_link_libraries(theone_core_tests PRIVATE one_content one_sample)
add_test(NAME core COMMAND theone_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_test(NAME convert_obj COMMAND ${CMAKE_COMMAND}
  -DIMPORTER=$<TARGET_FILE:one_import> -DINSPECTOR=$<TARGET_FILE:one_inspect>
  -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/assets/conversion/cube.obj -DDEST=${CMAKE_CURRENT_BINARY_DIR}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/conversion_test.cmake)
add_test(NAME vulkan_smoke COMMAND theone --smoke-test)
set_tests_properties(vulkan_smoke PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
get_filename_component(VULKAN_SDK_DIRECTORY "${Vulkan_INCLUDE_DIR}" DIRECTORY)
if(EXISTS "${VULKAN_SDK_DIRECTORY}/Bin/VkLayer_khronos_validation.json")
  add_test(NAME vulkan_validation COMMAND theone --smoke-test --validate)
  set_tests_properties(vulkan_validation PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE ENVIRONMENT "VK_LAYER_PATH=${VULKAN_SDK_DIRECTORY}/Bin")
endif()



if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/external/compatibility.one" AND EXISTS "${VULKAN_SDK_DIRECTORY}/Bin/VkLayer_khronos_validation.json")
  add_test(NAME sample_compatibility COMMAND theone --smoke-test --validate --scene "${CMAKE_CURRENT_BINARY_DIR}/external/compatibility.one")
  set_tests_properties(sample_compatibility PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE ENVIRONMENT "VK_LAYER_PATH=${VULKAN_SDK_DIRECTORY}/Bin")
endif()
