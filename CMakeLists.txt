cmake_minimum_required(VERSION 3.24)
project(TheOne VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(THEONE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
option(THEONE_BUILD_TOOLS "Build generic asset conversion tools" ${PROJECT_IS_TOP_LEVEL})
option(THEONE_BUILD_TESTS "Build engine-only regression tests" ${PROJECT_IS_TOP_LEVEL})
option(THEONE_BUILD_GPU_TESTS "Build GPU geometry regression and benchmark" ON)
include(cmake/Dependencies.cmake)
include(cmake/Shaders.cmake)

add_subdirectory(src/core)
add_subdirectory(src/animation)
add_subdirectory(src/geometry)
add_subdirectory(src/assets)
add_subdirectory(src/scene)
add_subdirectory(src/physics)
add_subdirectory(src/input)
add_subdirectory(src/effects)
add_subdirectory(src/lighting)
add_subdirectory(src/plasma)
add_subdirectory(src/volumes)
add_subdirectory(src/render)

# Explicit public modules. Existing Content consumers retain an aggregate compatibility target.
add_library(TheOne::Core ALIAS one_core)
add_library(TheOne::Animation ALIAS one_animation)
add_library(TheOne::Assets ALIAS one_assets)
add_library(TheOne::Scene ALIAS one_scene)
add_library(TheOne::Physics ALIAS one_physics)
add_library(TheOne::Input ALIAS one_input)
add_library(TheOne::Effects ALIAS one_effects)
add_library(TheOne::Renderer ALIAS one_renderer)

add_library(one_content INTERFACE)
target_link_libraries(one_content INTERFACE one_assets one_scene one_physics one_input one_effects nlohmann_json::nlohmann_json)
add_library(TheOne::Content ALIAS one_content)

if(THEONE_BUILD_TOOLS)
  add_subdirectory(tools)
endif()
if(THEONE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
